cmake_minimum_required(VERSION 3.16)
project(sf6_http_tests LANGUAGES CXX)

add_executable(store_check store_check.cpp ../src/store.cpp)
target_compile_features(store_check PRIVATE cxx_std_17)
target_include_directories(store_check PRIVATE ../third_party)

find_package(Threads REQUIRED)
target_link_libraries(store_check PRIVATE Threads::Threads)
add_executable(http_check http_check.cpp ../src/http.cpp)
target_compile_features(http_check PRIVATE cxx_std_17)
target_link_libraries(http_check PRIVATE Threads::Threads)
if(WIN32)
  target_compile_definitions(http_check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(http_check PRIVATE ws2_32)
endif()

add_executable(input_check input_check.cpp ../src/input.cpp)
target_compile_features(input_check PRIVATE cxx_std_17)
target_include_directories(input_check PRIVATE ../third_party)
target_link_libraries(input_check PRIVATE Threads::Threads)
if(WIN32)
  target_compile_definitions(input_check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(input_check PRIVATE xinput9_1_0 dinput8 dxguid ole32 user32)
endif()
